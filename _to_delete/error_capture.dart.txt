import 'package:flutter/material.dart';

/// TEMPORARY: captures the first framework error and shows it on-screen so it
/// can be read/copied without scrolling the console. Remove once diagnosed.
final ValueNotifier<String?> firstError = ValueNotifier<String?>(null);

void installErrorCapture() {
  final prev = FlutterError.onError;
  FlutterError.onError = (FlutterErrorDetails details) {
    if (firstError.value == null) {
      final frames =
          details.stack?.toString().split('\n').take(14).join('\n') ?? '';
      firstError.value =
          '${details.exceptionAsString()}\nlibrary: ${details.library ?? '-'}\n$frames';
    }
    prev?.call(details);
  };
}

class DebugErrorOverlay extends StatelessWidget {
  const DebugErrorOverlay({super.key, required this.child});
  final Widget child;

  @override
  Widget build(BuildContext context) {
    return Stack(
      textDirection: TextDirection.ltr,
      children: [
        child,
        Positioned(
          left: 0,
          right: 0,
          top: 0,
          child: ValueListenableBuilder<String?>(
            valueListenable: firstError,
            builder: (context, err, _) {
              if (err == null) return const SizedBox.shrink();
              return Material(
                color: const Color(0xFFB00020),
                child: SafeArea(
                  bottom: false,
                  child: Container(
                    constraints: const BoxConstraints(maxHeight: 260),
                    padding: const EdgeInsets.all(10),
                    child: Row(
                      crossAxisAlignment: CrossAxisAlignment.start,
                      children: [
                        Expanded(
                          child: SingleChildScrollView(
                            child: SelectableText(
                              err,
                              style: const TextStyle(color: Colors.white, fontSize: 10, height: 1.35),
                            ),
                          ),
                        ),
                        IconButton(
                          icon: const Icon(Icons.close, color: Colors.white, size: 18),
                          onPressed: () => firstError.value = null,
                        ),
                      ],
                    ),
                  ),
                ),
              );
            },
          ),
        ),
      ],
    );
  }
}
